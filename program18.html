PROGRAM 18: RUNGE KUTTA SECOND ORDER METHOD

C Program

#include<stdio.h> #define f(x,y) (x*y) void main() { float x0,y0,xn,h,yn,k1,k2,k; int i,n; printf("Enter Initial Condition\n"); printf("x0 = "); scanf("%f",&x0); printf("y0 = "); scanf("%f",&y0); printf("Enter calculation point xn = "); scanf("%f",&xn); printf("Enter number of steps: "); scanf("%d",&n); h=(xn-x0)/n; printf("\nx0\ty0\tyn\n"); for(i=0;i<n;i++) { k1=h*f(x0,y0); k2=h*f((x0+h),(y0+k1)); k=(k1+k2)/2; yn=y0+k; printf("%.4f\t%.4f\t%.4f\n",x0,y0,yn); x0=x0+h; y0=yn; } printf("\nValue of y at x = %.2f is %.3f",xn,yn); return; }

Example Input

Enter Initial Condition x0 = 0 y0 = 1 Enter calculation point xn = 1 Enter number of steps: 5

Output

x0 y0 yn 0.0000 1.0000 1.0200 0.2000 1.0200 1.0828 0.4000 1.0828 1.1963 0.6000 1.1963 1.3753 0.8000 1.3753 1.6448 Value of y at x = 1.00 is 1.645
