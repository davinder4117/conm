PROGRAM 8: NEWTON FORWARD DIFFERENCES TABLE

C Program

#include<stdio.h> #include<conio.h> void main() { int i,j,n; float x[20],y[20][20]; clrscr(); printf("\nEnter the value of N:"); scanf("%d",&n); printf("\nEnter pairs of (x,y):\n"); for(i=0;i<n;i++) { scanf("%f%f",&x[i],&y[i][0]); } for(j=1;j<n;j++) { for(i=0;i<n-j;i++) { y[i][j]=y[i+1][j-1]-y[i][j-1]; } } printf("\nNEWTON FORWARD DIFFERENCE TABLE\n"); printf("\nX\tY"); for(i=1;i<n;i++) { printf("\tD%d",i); } for(i=0;i<n;i++) { printf("\n%f",x[i]); for(j=0;j<n-i;j++) { printf("\t%f",y[i][j]); } } getch(); }

Example Input

Enter the value of N: 4 Enter pairs of (x,y): 0 1 1 2 2 5 3 10

Output

NEWTON FORWARD DIFFERENCE TABLE X Y D1 D2 D3 0 1 1 2 0 1 2 3 2 2 5 5 3 10
