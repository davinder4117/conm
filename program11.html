PROGRAM 11: NEWTON DIVIDED DIFFERENCE INTERPOLATION

C Program

#include<stdio.h> #define MAX 20 int main() { int i,j,k,n; float x[MAX],d[MAX][MAX],a,sum,product; printf("Enter the value of N: "); scanf("%d",&n); if(n<2 || n>MAX) return 0; printf("Enter pairs of (x,y):\n"); for(i=0;i<n;i++) { scanf("%f%f",&x[i],&d[i][0]); } printf("Enter the value of X: "); scanf("%f",&a); if(a<x[0] || a>x[n-1]) { printf("Value lies outside the tabulated range."); return 0; } for(j=1;j<n;j++) { for(i=0;i<n-j;i++) { d[i][j]=(d[i+1][j-1]-d[i][j-1])/(x[i+j]-x[i]); } } sum=d[0][0]; for(k=1;k<n;k++) { product=1.0; for(j=0;j<k;j++) { product=product*(a-x[j]); } sum=sum+d[0][k]*product; } printf("\nInterpolated value of Y = %f",sum); return 0; }

Example Input

Enter the value of N: 4 Enter pairs of (x,y): 0 1 1 2 2 9 3 28 Enter the value of X: 1.5

Output

Interpolated value of Y = 4.375000
