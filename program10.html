PROGRAM 10: NEWTON DIVIDED DIFFERENCES TABLE

C Program

#include<stdio.h> #define MAX 20 int main() { int i,j,n; float x[MAX],d[MAX][MAX]; printf("Enter the value of N: "); scanf("%d",&n); if(n<2 || n>MAX) return 0; printf("Enter pairs of (x,y):\n"); for(i=0;i<n;i++) { scanf("%f%f",&x[i],&d[i][0]); } for(j=1;j<n;j++) { for(i=0;i<n-j;i++) { d[i][j]=(d[i+1][j-1]-d[i][j-1])/(x[i+j]-x[i]); } } printf("\nNewton Divided Difference Table\n"); printf("x\tf[x]"); for(j=1;j<n;j++) { printf("\tD%d",j); } printf("\n"); for(i=0;i<n;i++) { printf("%.2f\t",x[i]); for(j=0;j<n-i;j++) { printf("%.2f\t",d[i][j]); } printf("\n"); } return 0; }

Example Input

Enter the value of N: 4 Enter pairs of (x,y): 0 1 1 2 2 9 3 28

Output

Newton Divided Difference Table x f[x] D1 D2 D3 0.00 1.00 1.00 3.00 1.00 1.00 2.00 7.00 6.00 2.00 9.00 19.00 3.00 28.00
