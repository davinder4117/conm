PROGRAM 15: SIMPSON'S 3/8 RULE

C Program

#include<stdio.h> #define MAX 50 float f(float x) { return 1.0/(1.0+x*x); } int main() { float a,b,h; float x[MAX],y[MAX]; float sum,intg; int i,n; printf("Enter values of a and b: "); scanf("%f%f",&a,&b); printf("Enter number of sub-intervals: "); scanf("%d",&n); if(a>b || n<3 || n%3!=0 || n>MAX-1) { printf("Invalid Input. Number of sub-intervals must be a multiple of 3."); return 0; } h=(b-a)/n; for(i=0;i<=n;i++) { x[i]=a+i*h; y[i]=f(x[i]); } sum=y[0]+y[n]; for(i=1;i<n;i++) { if(i%3==0) { sum=sum+2*y[i]; } else { sum=sum+3*y[i]; } } intg=(3*h/8)*sum; printf("Integral = %.4f\n",intg); return 0; }

Example Input

Enter values of a and b: 0 1 Enter number of sub-intervals: 3

Output

Integral = 0.7846
