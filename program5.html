PROGRAM 5: GAUSS ELIMINATION METHOD

C Program

#include<stdio.h> #include<conio.h> #include<stdlib.h> #define max 50 void main() { int i,j,k,n; float a[max][max],x[max],u,sum; clrscr(); printf("\nEnter the number of equations:"); scanf("%d",&n); for(i=0;i<n;i++) { printf("\nEnter the coefficients of unknowns and RHS value of equation %d:",i+1); for(j=0;j<=n;j++) { scanf("%f",&a[i][j]); } } for(k=0;k<n-1;k++) { if(a[k][k]==0) { printf("\nDivision by zero"); exit(0); } for(i=k+1;i<n;i++) { u=a[i][k]/a[k][k]; for(j=0;j<=n;j++) { a[i][j]=a[i][j]-u*a[k][j]; } } } for(i=n-1;i>=0;i--) { sum=0; for(j=i+1;j<n;j++) { sum+=a[i][j]*x[j]; } x[i]=(a[i][n]-sum)/a[i][i]; } printf("\nSolution of system linear equations is:\n"); for(i=0;i<n;i++) { printf("\n x[%d]=%f",i+1,x[i]); } getch(); }

Example Input

Enter the number of equations: 2 Enter the coefficients of unknowns and RHS value of equation 1: 2 1 5 Enter the coefficients of unknowns and RHS value of equation 2: 1 3 6

Output

Solution of system linear equations is: x[1]=1.800000 x[2]=1.400000
