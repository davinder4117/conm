#include<stdio.h> #include<conio.h> #include<math.h> #include<stdlib.h> void main() { float a[20],b[20],c[20],x,epsilon,xnew; int n,i,k,maxitr; clrscr(); printf("Enter the degree of polynomial:\n"); scanf("%d",&n); printf("Enter the initial approximation to root:\n"); scanf("%f",&x); printf("Enter the xmpscribed tolerance in the root:\n"); scanf("%f",&epsilon); printf("Enter the maximum number of iterations:\n"); scanf("%d",&maxitr); printf("Enter the coefficients of polynomial:\n"); for(i=0;i<=n;i++) { scanf("%f",&a[i]); } for(k=1;k<=maxitr;k++) { b[0]=a[0]; for(i=1;i<=n;i++) { b[i]=a[i]+x*b[i-1]; } c[0]=b[0]; for(i=1;i<n;i++) { c[i]=b[i]+x*c[i-1]; } if(c[n-1]==0) { printf("Division by zero"); exit(0); } xnew=x-b[n]/c[n-1]; if(fabs(xnew-x)<=epsilon) { printf("Root is %f",xnew); getch(); exit(0); } x=xnew; } printf("Root not found in %d iterations",maxitr); getch(); }
