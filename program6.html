PROGRAM 6: GAUSS JORDAN METHOD

C Program

#include<stdio.h> #include<conio.h> #include<stdlib.h> #define max 50 void main() { int i,j,k,n; float a[max][max],temp; clrscr(); printf("\nEnter the number of equations:"); scanf("%d",&n); for(i=0;i<n;i++) { printf("\nEnter the coefficients of unknowns and RHS value of equation %d:",i+1); for(j=0;j<=n;j++) { scanf("%f",&a[i][j]); } } for(k=0;k<n;k++) { if(a[k][k]==0) { printf("\nDivision by zero"); exit(0); } temp=a[k][k]; for(j=0;j<=n;j++) { a[k][j]=a[k][j]/temp; } for(i=0;i<n;i++) { if(i!=k) { temp=a[i][k]; for(j=0;j<=n;j++) { a[i][j]=a[i][j]-temp*a[k][j]; } } } } printf("\nSolution of system of linear equations is:\n"); for(i=0;i<n;i++) { printf("\nx[%d]=%f",i+1,a[i][n]); } getch(); }

Example Input

Enter the number of equations: 2 Enter the coefficients of unknowns and RHS value of equation 1: 2 1 5 Enter the coefficients of unknowns and RHS value of equation 2: 1 3 6

Output

Solution of system of linear equations is: x[1]=1.800000 x[2]=1.400000
