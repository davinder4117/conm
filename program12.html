PROGRAM 12: LAGRANGE INTERPOLATION POLYNOMIAL

C Program

#include<stdio.h> int main() { int n; int i,j; double sum=0.0; double prod; double xp; printf("Enter number of data points: "); scanf("%d",&n); if(n<2) { printf("Invalid number of data points."); return 0; } double x[n]; double y[n]; printf("Enter the x and y values:\n"); for(i=0;i<n;i++) { printf("x[%d]: ",i); scanf("%lf",&x[i]); printf("y[%d]: ",i); scanf("%lf",&y[i]); } printf("Enter the value of x for interpolation: "); scanf("%lf",&xp); for(i=0;i<n;i++) { prod=1.0; for(j=0;j<n;j++) { if(i!=j) { prod=prod*(xp-x[j])/(x[i]-x[j]); } } sum=sum+y[i]*prod; } printf("Interpolated value = %.3lf\n",sum); return 0; }

Example Input

Enter number of data points: 4 Enter the x and y values: x[0]: 0 y[0]: 1 x[1]: 1 y[1]: 2 x[2]: 2 y[2]: 9 x[3]: 3 y[3]: 28 Enter the value of x for interpolation: 1.5

Output

Interpolated value = 4.375
