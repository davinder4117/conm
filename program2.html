PROGRAM 2: FALSE POSITION METHOD

C Program

#include<stdio.h> #include<conio.h> #include<math.h> #include<stdlib.h> #define f(x) (x*x*x-x-4) void main() { int k=0; float f1,f2,f3,x1,x2,x3,epsilon,delta; clrscr(); printf("\nEnter two initial approximations x1,x2: "); scanf("%f,%f",&x1,&x2); printf("\nEnter very small number epsilon and delta: "); scanf("%f,%f",&epsilon,&delta); f1=f(x1); f2=f(x2); if(f1*f2>0) { printf("\nInitial approximations are not proper"); exit(0); } do { if(fabs(f2-f1)<delta) { printf("\nSlope curve is too small"); exit(0); } x3=(x1*f2-x2*f1)/(f2-f1); k++; printf("\nNext approximation after %d iteration is %f",k,x3); f3=f(x3); if(f1*f3>0) { x2=x3; f2=f3; } else { x1=x3; f1=f3; } }while(fabs((x2-x1)/x2)>epsilon && f3!=0); printf("\nRoot=%10.4f",x3); getch(); }
