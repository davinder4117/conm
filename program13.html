PROGRAM 13: TRAPEZOIDAL RULE

C Program

#include<stdio.h> float f(float x) { return 1.0/(1.0+x*x*x*x); } int main() { int n; int i; float a,b,h; float sum,intg; printf("Enter values of a and b: "); scanf("%f%f",&a,&b); printf("Enter number of sub-intervals: "); scanf("%d",&n); if(a>b || n<1) { printf("Invalid Input."); return 0; } h=(b-a)/n; sum=f(a)+f(b); for(i=1;i<n;i++) { sum=sum+2*f(a+i*h); } intg=(h/2)*sum; printf("Integral = %.4f\n",intg); return 0; }

Example Input

Enter values of a and b: 0 1 Enter number of sub-intervals: 4

Output

Integral = 0.8617
