PROGRAM 3: NEWTON RAPHSON METHOD

C Program

#include<stdio.h> #include<conio.h> #include<math.h> #include<stdlib.h> #define f(x) (x*x*x-x-4) #define df(x) (3*x*x-1) void main() { int k,n; float x1,x2,epsilon,delta,err; clrscr(); printf("\nEnter the value of x1: "); scanf("%f",&x1); printf("\nEnter the values of epsilon and delta: "); scanf("%f,%f",&epsilon,&delta); printf("\nEnter the number of iteration: "); scanf("%d",&n); for(k=1;k<=n;k++) { if(fabs(df(x1))<delta) { printf("\nSlope of curve is too small"); exit(0); } x2=x1-f(x1)/df(x1); err=fabs((x2-x1)/x2); if(err<epsilon) { printf("\nRoot=%f",x2); exit(0); } x1=x2; } printf("\nSolution does not converge after %d iterations",n); getch(); }

Example Input

Enter the value of x1: 2 Enter the values of epsilon and delta: 0.001,0.0001 Enter the number of iteration: 20

Output

Root = 1.796322
