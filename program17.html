PROGRAM 17: MODIFIED EULER'S METHOD

C Program

#include<stdio.h> #include<math.h> double f(double x,double y) { double y_dash; y_dash=x*x+y; return(y_dash); } void main() { double x0,y0,y1,y10,x1,h,x,diff; int i; printf("\nMODIFIED EULER'S METHOD TO SOLVE DIFFERENTIAL EQUATION"); printf("\nEnter x0 = "); scanf("%lf",&x0); printf("\nEnter y0 = "); scanf("%lf",&y0); printf("\nEnter the value of x at which y is to be found = "); scanf("%lf",&x); printf("\nEnter the value of h = "); scanf("%lf",&h); i=0; while(x0<x) { i++; x1=x0+h; y1=y0+h*f(x0,y0); y10=y1; do { y1=y0+(h/2)*(f(x0,y0)+f(x1,y10)); diff=fabs(y1-y10); y10=y1; }while(diff>0.0001); printf("\n%d = %lf y%d = %lf",i,x1,i,y1); x0=x1; y0=y1; } }

Example Input

Enter x0 = 0 Enter y0 = 1 Enter the value of x at which y is to be found = 1 Enter the value of h = 0.2
