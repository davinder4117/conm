PROGRAM 7: GAUSS SEIDEL METHOD

C Program

#include<stdio.h> #include<conio.h> #include<math.h> void main() { int n,maxiter,i,j,k; float a[10][11],x[10],sum,temp,error,epsilon,bigerror; clrscr(); printf("\nEnter number of equations:"); scanf("%d",&n); for(i=0;i<n;i++) { printf("\nEnter coefficients of equation %d:",i+1); for(j=0;j<=n;j++) { scanf("%f",&a[i][j]); } } printf("\nEnter maximum number of iterations:"); scanf("%d",&maxiter); printf("\nEnter a very small number epsilon:"); scanf("%f",&epsilon); for(i=0;i<n;i++) { x[i]=0; } for(k=1;k<=maxiter;k++) { bigerror=0; for(i=0;i<n;i++) { sum=0; for(j=0;j<n;j++) { if(i!=j) { sum+=a[i][j]*x[j]; } } temp=(a[i][n]-sum)/a[i][i]; error=fabs((temp-x[i])/temp); if(error>bigerror) { bigerror=error; } x[i]=temp; } if(bigerror<epsilon) { break; } } printf("\nSolution of system of linear equations is:\n"); for(i=0;i<n;i++) { printf("\nx[%d]=%f",i+1,x[i]); } getch(); }

Example Input

Enter number of equations: 2 Enter coefficients of equation 1: 4 1 9 Enter coefficients of equation 2: 1 3 5 Enter maximum number of iterations: 20 Enter a very small number epsilon: 0.001

Output

Solution of system of linear equations is: x[1]=2.000000 x[2]=1.000000
